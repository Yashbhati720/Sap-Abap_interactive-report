*&---------------------------------------------------------------------*
*& Report ZINTERACTIVE_REPORT
*& Description: 3-Level Interactive Basic List (PO Header -> Item -> History)
*&---------------------------------------------------------------------*
REPORT zinteractive_report.

TABLES: ekko, 
        ekpo.

*----------------------------------------------------------------------*
* Data Declarations
*----------------------------------------------------------------------*
DATA: gt_ekko TYPE TABLE OF ekko,
      gs_ekko TYPE ekko,
      gt_ekpo TYPE TABLE OF ekpo,
      gs_ekpo TYPE ekpo,
      gt_ekbe TYPE TABLE OF ekbe,
      gs_ekbe TYPE ekbe.

DATA: gv_ebeln TYPE ekko-ebeln,
      gv_ebelp TYPE ekpo-ebelp.

*----------------------------------------------------------------------*
* Selection Screen
*----------------------------------------------------------------------*
SELECTION-SCREEN BEGIN OF BLOCK b1 WITH FRAME TITLE TEXT-001.
  SELECT-OPTIONS: s_ebeln FOR ekko-ebeln,
                  s_bukrs FOR ekko-bukrs,
                  s_werks FOR ekpo-werks,
                  s_aedat FOR ekko-aedat,
                  s_lifnr FOR ekko-lifnr.
SELECTION-SCREEN END OF BLOCK b1.

*----------------------------------------------------------------------*
* INITIALIZATION
*----------------------------------------------------------------------*
INITIALIZATION.
  s_aedat-sign   = 'I'.
  s_aedat-option = 'BT'.
  s_aedat-low    = sy-datum - 365.
  s_aedat-high   = sy-datum.
  APPEND s_aedat.

*----------------------------------------------------------------------*
* START-OF-SELECTION : Fetch Level 1 (PO Header Data)
*----------------------------------------------------------------------*
START-OF-SELECTION.

  SELECT * FROM ekko
    INTO TABLE gt_ekko
    WHERE ebeln IN s_ebeln
      AND bukrs IN s_bukrs
      AND aedat IN s_aedat
      AND lifnr IN s_lifnr.

  IF gt_ekko IS INITIAL.
    MESSAGE 'No purchase orders found for selection' TYPE 'I'.
  ENDIF.

*----------------------------------------------------------------------*
* END-OF-SELECTION : Render Level 1 List
*----------------------------------------------------------------------*
END-OF-SELECTION.

  PERFORM display_header_list.

*----------------------------------------------------------------------*
* TOP-OF-PAGE : Standard Header for Lists
*----------------------------------------------------------------------*
TOP-OF-PAGE.
  WRITE: / 'Interactive Purchase Report', sy-datum, sy-uzeit.
  ULINE.

*----------------------------------------------------------------------*
* AT LINE-SELECTION : Triggered on Row Click/Double-Click
*----------------------------------------------------------------------*
AT LINE-SELECTION.

  CASE sy-lsind.
    WHEN 1.
      " Coming from Level 1 (Header list) -> show items
      gv_ebeln = gs_ekko-ebeln.
      PERFORM get_and_display_items.

    WHEN 2.
      " Coming from Level 2 (Item list) -> show GR/IR history
      PERFORM get_and_display_history.
  ENDCASE.

*&---------------------------------------------------------------------*
*& Form DISPLAY_HEADER_LIST (Level 1)
*&---------------------------------------------------------------------*
FORM display_header_list.

  WRITE: / 'Purchase Order Overview' COLOR COL_HEADING,
         / sy-uline.
  WRITE: / 'PO Number'   COLOR COL_HEADING,
           'Vendor'      COLOR COL_HEADING,
           'Doc. Date'   COLOR COL_HEADING,
           'Company'     COLOR COL_HEADING,
           'Currency'    COLOR COL_HEADING.
  ULINE.

  LOOP AT gt_ekko INTO gs_ekko.
    HIDE gs_ekko. " Store record details for line selection
    WRITE: / gs_ekko-ebeln COLOR COL_KEY HOTSPOT ON,
             gs_ekko-lifnr,
             gs_ekko-aedat,
             gs_ekko-bukrs,
             gs_ekko-waers.
  ENDLOOP.

  WRITE: / sy-uline.
  WRITE: / 'Double-click / click on a PO number to see items'.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form GET_AND_DISPLAY_ITEMS (Level 2)
*&---------------------------------------------------------------------*
FORM get_and_display_items.

  CLEAR gt_ekpo.
  SELECT * FROM ekpo
    INTO TABLE gt_ekpo
    WHERE ebeln = gv_ebeln
      AND werks IN s_werks.

  IF gt_ekpo IS INITIAL.
    MESSAGE 'No items found for this PO' TYPE 'I'.
    EXIT.
  ENDIF.

  WRITE: / 'Items for PO:', gv_ebeln COLOR COL_POSITIVE.
  WRITE: / sy-uline.
  WRITE: / 'Item'      COLOR COL_HEADING,
           'Material'  COLOR COL_HEADING,
           'Plant'     COLOR COL_HEADING,
           'Qty'       COLOR COL_HEADING,
           'UoM'       COLOR COL_HEADING,
           'Net Price' COLOR COL_HEADING,
           'Deleted'   COLOR COL_HEADING.
  ULINE.

  LOOP AT gt_ekpo INTO gs_ekpo.
    HIDE gs_ekpo.
    WRITE: / gs_ekpo-ebelp COLOR COL_KEY HOTSPOT ON,
             gs_ekpo-matnr,
             gs_ekpo-werks,
             gs_ekpo-menge UNIT gs_ekpo-meins,
             gs_ekpo-meins,
             gs_ekpo-netpr CURRENCY gs_ekko-waers,
             gs_ekpo-loekz AS CHECKBOX.
  ENDLOOP.

  WRITE: / sy-uline.
  WRITE: / 'Click an item line to see GR / Invoice history'.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form GET_AND_DISPLAY_HISTORY (Level 3)
*&---------------------------------------------------------------------*
FORM get_and_display_history.

  gv_ebelp = gs_ekpo-ebelp.

  CLEAR gt_ekbe.
  SELECT * FROM ekbe
    INTO TABLE gt_ekbe
    WHERE ebeln = gv_ebeln
      AND ebelp = gv_ebelp.

  IF gt_ekbe IS INITIAL.
    MESSAGE 'No GR/Invoice history for this item' TYPE 'I'.
    EXIT.
  ENDIF.

  WRITE: / 'GR / Invoice history for', gv_ebeln, gv_ebelp COLOR COL_POSITIVE.
  WRITE: / sy-uline.
  WRITE: / 'Doc. No.'     COLOR COL_HEADING,
           'Doc. Year'    COLOR COL_HEADING,
           'Mvt Type'     COLOR COL_HEADING,
           'Posting Date' COLOR COL_HEADING,
           'Qty'          COLOR COL_HEADING,
           'Amount'       COLOR COL_HEADING.
  ULINE.

  LOOP AT gt_ekbe INTO gs_ekbe.
    WRITE: / gs_ekbe-belnr,
             gs_ekbe-gjahr,
             gs_ekbe-bwart,
             gs_ekbe-budat,
             gs_ekbe-menge UNIT gs_ekpo-meins,
             gs_ekbe-dmbtr CURRENCY gs_ekko-waers.
  ENDLOOP.

  WRITE: / sy-uline.

ENDFORM.
